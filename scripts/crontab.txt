# ------------------------------------------------------------------------------
# รูปแบบที่ 1: เซิร์ฟเวอร์ VPS ตั้งเวลาเป็น Asia/Bangkok แล้ว (แนะนำวิธีนี้)
# ตรวจสอบเวลาด้วยคำสั่ง: date
# หากยังไม่เป็นเวลาไทย ตั้งค่าง่ายๆ ด้วยคำสั่ง: sudo timedatectl set-timezone Asia/Bangkok
# ------------------------------------------------------------------------------
CRON_TZ=Asia/Bangkok

# 1. รอบเช้า 07:45 น. (จันทร์ - ศุกร์): สรุปข่าวทองคำ XAU/USD, หุ้นไทยเด่น, Sector Rotation
45 7 * * 1-5 /bin/bash /root/onicorn-trade/scripts/run_vps_alert.sh morning >> /var/log/onicorn_alerts.log 2>&1

# 2. ภาพกราฟสดน้ำมัน WTI 08:00 น. (ทุกวันอังคาร): แคปเจอร์กราฟ TradingView + ราคาน้ำมันขายปลีกไทย
0 8 * * 2 /bin/bash /root/onicorn-trade/scripts/run_vps_alert.sh oil_chart >> /var/log/onicorn_alerts.log 2>&1

# 3. หุ้นทะลุกรอบ & วอลุ่มพุ่ง 14:35 น. (จันทร์ - ศุกร์): ตรวจจับหุ้นเบรกเอาท์ระหว่างวัน (ไม่ส่งหากไม่มีหุ้นเข้าเกณฑ์)
35 14 * * 1-5 /bin/bash /root/onicorn-trade/scripts/run_vps_alert.sh intraday >> /var/log/onicorn_alerts.log 2>&1

# 4. สรุปตลาดสิ้นวัน 17:20 น. (จันทร์ - ศุกร์): รายงานดัชนี SET และมูลค่าซื้อขายแยกตามกลุ่มนักลงทุน
20 17 * * 1-5 /bin/bash /root/onicorn-trade/scripts/run_vps_alert.sh market_wrap >> /var/log/onicorn_alerts.log 2>&1

# ------------------------------------------------------------------------------
# รูปแบบที่ 2: กรณีเซิร์ฟเวอร์ VPS เป็นเวลาสากล UTC (ไม่ได้เปลี่ยน Timezone)
# เวลาไทย = UTC + 7 ชั่วโมง
# ------------------------------------------------------------------------------
# 00:45 UTC = 07:45 น. (จันทร์ - ศุกร์) -> สรุปข่าวทองคำ & หุ้นเด่น
# 45 0 * * 1-5 /bin/bash /root/onicorn-trade/scripts/run_vps_alert.sh morning >> /var/log/onicorn_alerts.log 2>&1
#
# 01:00 UTC = 08:00 น. (วันอังคาร) -> ภาพกราฟสดน้ำมัน WTI
# 0 1 * * 2 /bin/bash /root/onicorn-trade/scripts/run_vps_alert.sh oil_chart >> /var/log/onicorn_alerts.log 2>&1
#
# 07:35 UTC = 14:35 น. (จันทร์ - ศุกร์) -> หุ้นทะลุกรอบ Intraday
# 35 7 * * 1-5 /bin/bash /root/onicorn-trade/scripts/run_vps_alert.sh intraday >> /var/log/onicorn_alerts.log 2>&1
#
# 10:20 UTC = 17:20 น. (จันทร์ - ศุกร์) -> สรุปตลาดสิ้นวัน Market Wrap
# 20 10 * * 1-5 /bin/bash /root/onicorn-trade/scripts/run_vps_alert.sh market_wrap >> /var/log/onicorn_alerts.log 2>&1
